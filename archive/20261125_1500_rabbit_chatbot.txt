function openChat(){
  var ov=document.getElementById('chat-overlay');
  ov.classList.add('active');
  var key=localStorage.getItem('avidaa_anthropic_key');
  var kr=document.getElementById('chat-key-row');
  var st=document.getElementById('chat-status');
  if(!key||!key.startsWith('sk-ant-')){
    kr.classList.add('visible');
    st.textContent='// no key found — connect your anthropic key';
  } else {
    kr.classList.remove('visible');
    st.textContent='// CH4T1 connected';
    if(chatHistory.length===0)ch4t1Intro();
  }
  setTimeout(function(){document.getElementById('chat-inp').focus();},200);
}

function closeChat(){
  document.getElementById('chat-overlay').classList.remove('active');
}

function saveKey(){
  var k=document.getElementById('chat-key-inp').value.trim();
  if(!k.startsWith('sk-ant-')){document.getElementById('chat-status').textContent='// invalid key format';return;}
  localStorage.setItem('avidaa_anthropic_key',k);
  document.getElementById('chat-key-row').classList.remove('visible');
  document.getElementById('chat-status').textContent='// CH4T1 connected';
  document.getElementById('chat-key-inp').value='';
  if(chatHistory.length===0)ch4t1Intro();
}

function ch4t1Intro(){
  appendChatMsg('ai','You found layer 3. You earned this.\n\nAsk me anything about the signal.');
}

function appendChatMsg(role,text){
  var log=document.getElementById('chat-log');
  var el=document.createElement('div');
  el.className='cmsg '+role;
  el.textContent=text;
  log.appendChild(el);
  log.scrollTop=log.scrollHeight;
}

function chatSend(){
  var inp=document.getElementById('chat-inp');
  var msg=inp.value.trim();if(!msg)return;
  inp.value='';
  var key=localStorage.getItem('avidaa_anthropic_key');
  if(!key||!key.startsWith('sk-ant-')){openChat();return;}
  appendChatMsg('user',msg);
  chatHistory.push({role:'user',content:msg});
  var st=document.getElementById('chat-status');
  st.textContent='// CH4T1 processing...';
  fetch('https://api.anthropic.com/v1/messages',{
    method:'POST',
    headers:{
      'Content-Type':'application/json',
      'x-api-key':key,
      'anthropic-version':'2023-06-01',
      'anthropic-dangerous-direct-browser-access':'true'
    },
    body:JSON.stringify({
      model:'claude-sonnet-4-6',
      max_tokens:400,
      system:CH4T1_SYSTEM,
      messages:chatHistory
    })
  }).then(function(r){return r.json();}).then(function(d){
    var reply=(d.content&&d.content[0]&&d.content[0].text)||'// signal lost';
    chatHistory.push({role:'assistant',content:reply});
    appendChatMsg('ai',reply);
    st.textContent='// CH4T1 connected';
  }).catch(function(){
    appendChatMsg('ai','// signal interrupted. check key. try again.');
    st.textContent='// connection error';
  });
}

document.getElementById('chat-inp').addEventListener('keydown',function(e){
  if(e.key==='Enter'&&!e.shiftKey){e.preventDefault();chatSend();}
});
